# Manual-install reference.
#
# These steps are now AUTOMATED by scripts/bootstrap_terminal.sh (run as part
# of `make mac_setup`). Kept here as documentation / manual fallback.

# Oh My Zsh + zsh-autosuggestions
sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" ;
git clone https://github.com/zsh-users/zsh-autosuggestions ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions

# Oh My Tmux (gpakosz) + tpm. Our edits live in ~/.tmux.conf.local (deployed by
# the Makefile); ~/.tmux.conf must stay a symlink to the framework.
cd ;
git clone https://github.com/gpakosz/.tmux.git ;
ln -s -f .tmux/.tmux.conf ;
git clone https://github.com/tmux-plugins/tpm ~/.tmux/plugins/tpm
# then install plugins: ~/.tmux/plugins/tpm/bin/install_plugins  (or prefix+I)

# Neovim plugins/parsers install automatically on first launch (lazy.nvim +
# treesitter main branch). The bootstrap script pre-warms them headlessly.

# Legacy Vim (vim-plug) — only if you still use the vim/ config:
curl -fLo ~/.vim/autoload/plug.vim --create-dirs \
    https://raw.githubusercontent.com/junegunn/vim-plug/master/plug.vim
